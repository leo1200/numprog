\section{Solving Linear Systems}
\label{sec:linear}
\thispagestyle{plain}

Solving linear systems $\mat{A}\vec{x} = \vec{b}$ (invertible $\mat{A} \in \mathbb{R}^{n \times n}, \vec{x} \in \mathbb{R}^{n}, \vec{b} \in \mathbb{R}^{n}$) 
is important throughout scientific computing and at the core of many numerical algorithms.

Numerical methods for solving linear systems include

\begin{itemize}
    \item \textbf{Direct solvers: Gaussian elimination / factorization methods}: The linear system is transformed into a 
    triangular form which can be trivially solved. Reaching the triangular form has a cost of $\mathcal{O}(n^3)$
    operations which can be prohibitive for large matrices.
    \item \textbf{Iterative methods}: An iterative procedure is designed which converges to the solution
    of the linear system (a dynamical system with $\vec{x}$ as the steady state).
    \begin{itemize}
        \item Classical iterative schemes include Jacobi and Gauss-Seidel iteration. 
        \item An important class of iterative methods are Krylov subspace methods (like \textit{GMRES}) which do not require $\mat{A}$ to be 
        materialized, only matrix vector products $\mat{A}\vec{v}$ are necessary. This is for instance very helpful 
        when $\mat{A}$ is a prohibitively large Jacobian.
    \end{itemize}
    For large, especially for sparse, matrices, iterative methods typically outperform direct solvers.
\end{itemize}

We will cover Gaussian elimination, LU factorization and classical iterative schemes here, in the appendix
you can find Krylov subspace methods (Section \ref{app:krylov}) and 
multigrid techniques (Section \ref{app:multigrid}).

\subsection{Why don't we typically explicitly invert matrices?}
One can efficiently invert matrices numerically. Indeed, matrix multiplication and 
inversion are equivalently hard problems in terms of asymptotic runtime \citep[Sec. 28.2]{cormen}, 
such that the scaling of Strassen's algorithm, $\mathcal{O}(N^{\log_2 7})$ \citep[Sec. 4.2]{cormen}, applies.

However, while $\mat{A} \in \mathbb{R}^{N\times N}$ 
might often be sparse (e.g. a sparse Jacobian), $\mat{A}^{-1}$ is 
dense in general and our storage might not be able to handle $N^2$ 
terms of a dense inverse. \enquote{Note that some may say that this is done 
for numerical stability reasons: that is incorrect. In fact, 
under reasonable assumptions for how the inverse is computed,
it will be as numerically stable as other 
techniques we will mention.} \citep[Ch. 6]{rackauckas22}.

\subsection{Example application of solving linear systems - linear regression}
\label{sec:linear_regression}

Given $N$ measurements of $p$ different \textit{features} $x_{i1},...,x_{ip}$ (these might for
instance be concentrations of different drugs administered to a patient) and outputs $y_i^{\text{data}}$ 
(for instance the concentration of pathogens one day after drug administration), we want
to find parameters $\beta_0,...,\beta_p$ in the linear relation

\begin{equation}
    y_i^{\text{predicted}} = \beta_0 + \sum_{k=1}^{p} \beta_k x_{ik}
\end{equation}

such that $\sum_{i=1}^{N} ||y_i^{\text{predicted}} - y_i^{\text{data}}||^2$ is minimized.

In matrix notation with

\begin{equation}
 \begin{gathered}
    \vec{y}=\left(\begin{array}{c}
        y_1 \\
        \vdots \\
        y_i \\
        \vdots \\
        y_N
        \end{array}\right), \quad \mat{X}=\left[\begin{array}{cccc}
        1 & x_{11} & \cdots & x_{1 p} \\
        \vdots & \vdots & \ddots & \vdots \\
        1 & x_{N 1} & \cdots & x_{N p}
    \end{array}\right] \in \mathbb{R}^{N \times p} \\
    \mat{X} \text{ with 1 column to allow for constant offset } \beta_0
 \end{gathered}
\end{equation}

we can write our model as

\begin{equation}
    \vec{y}^{\text{predicted}} = \mat{X} \vec{\beta}
\end{equation}

We can therefore estimate the \textit{optimal} parameters with

\begin{equation}
    \hat{\vec{\beta}} = \argmin_{\vec{\beta}}{||\vec{y}^{\text{data}} - \mat{X} \vec{\beta}||_2^2}
\end{equation}

so

\begin{equation}
    \partial_{\vec{\beta}} ||\vec{y}^{\text{data}} - \mat{X} \vec{\beta}||_2^2 = -2 \mat{X}^T(\vec{y}^{\text{data}}-\mat{X} \vec{\beta}) \underbracket{=}_{!} 0
\end{equation}

and written as a linear system

\begin{equation}
   \mat{X}^T \mat{X} \vec{\beta} = \mat{X}^T \vec{y}^{\text{data}}
\end{equation}

\subsection{Direct solvers: Gaussian elimination and LU factorization}

\subsubsection{Introduction on an example}

\begin{equation}
    \begin{aligned}

        x_1 - 2x_2 = 4

\end{equation}

We can eliminate $x_1$ from the second equation by subtracting the first
equation, such that we obtain

\begin{equation}

\subsubsection{General principle}

Formulated abstractly, we

\begin{enumerate}
    \item go through the columns of $\mat{A}$ from left to 
    right and eliminate all entries below the diagonal until we obtain an upper 
    system, at the $k$-th stage of this procedure, we have reached a system $\mat{A}^{(k)}\vec{x} = b^{(k)}$
    where
    \begin{equation}
        \mat{A}^{(k)} = \begin{pmatrix}
            \mat{A}_{11}^{(k)} & \mat{A}_{12}^{(k)} \\
            0 & \mat{A}_{22}^{(k)}
        \end{pmatrix}
    \end{equation}
    with $\mat{A}_{11}^{(k)} \in \mathcal{R}^{(k-1) \times (k-1)}$ upper triangular. We then eliminate
    all elements in the $k$-th column below the diagonal by subtracting scaled versions on the $k$-th row
    to all rows $i$ with $i>k$:
    \begin{equation}
        \begin{array}{ll}
            a_{i j}^{(k+1)}=a_{i j}^{(k)}-m_{i k} a_{k j}^{(k)}, & i=k+1, ..., n, j=k+1, ..., n, \\
            b_i^{(k+1)}=b_i^{(k)}-m_{i k} b_k^{(k)}, & i=k+1, ..., n,
        \end{array}
    \end{equation}
    with $m_{ik} = \frac{a^{(k)}_{ik}}{a^{(k)}_{kk}}$. Each stage requires $\mathcal{O}(n^2)$ operations, the whole procedure $\mathcal{O}(n^3)$.
    $j$ starts at $k+1$, because we already know the result for $j=k$ (by design $a_{ik}^{(k+1)} = 0,i>k$).
    \item the triangular system can then be solved bottom-up
\end{enumerate}

\subsubsection{The need for pivoting}

\redbox{\textbf{Problem:} The procedure above breaks down for $a^{(k)}_{kk} = 0$. For large $m_{ik} = \frac{a^{(k)}_{ik}}{a^{(k)}_{kk}}$, 
low-order digits of $a_{i j}^{(k)}$ and $b_i^{(k)}$ will be lost.}

Consider the simple example system

\begin{equation}
    \begin{pNiceArray}{cc|c}[margin]
        \epsilon & 1 & 1 \\
        1 & 1 & 2 \\
    \end{pNiceArray}
\end{equation}

with Gaussian elimination we yield

\begin{equation}
    \begin{pNiceArray}{cc|c}[margin]
        \epsilon & 1 & 1 \\
        0 & 1 - \frac{1}{\epsilon} & 2  - \frac{1}{\epsilon} \\
    \end{pNiceArray}
\end{equation}

breaking down for $\epsilon = 0$ and leading 
to cancellation for small $\epsilon$.

\greenbox{\textbf{Idea:} We exchange rows (and columns) to minimize $m_{ik}$ at the $k$-th
stage of Gaussian elimination. For instance
\begin{equation}
        \begin{pNiceArray}{cc|c}[margin]
        \epsilon & 1 & 1 \\
        1 & 1 & 2 \\
    \end{pNiceArray}
    \rightarrow
    \begin{pNiceArray}{cc|c}[margin]
        1 & 1 & 2 \\
        \epsilon & 1 & 1
    \end{pNiceArray}
    \rightarrow
    \begin{pNiceArray}{cc|c}[margin]
        1 & 1 & 2 \\
        0 & 1 - \epsilon & 1 - \epsilon
    \end{pNiceArray}
\end{equation}
is unproblematic.
}

\bluebox{\textbf{Partial pivoting}: At the start of the $k$-th stage of Gaussian elimination exchange the
$k$-th and $r$-th row, where
\begin{equation}
    r = \argmax_{k \leq i \leq n} |a_{ik}^{(k)}|
\end{equation}
such that $m_{ik} < 1, i=k+1, ..., n$.
}

One can also combine row and column swaps to minimize $m_{ik}$ further (\textit{complete pivoting}) but this
requires $\mathcal{O}(n^3)$ comparison operations in total compared to the $\mathcal{O}(n^2)$ comparison operations 
of partial pivoting.

The stability of Gaussian elimination with a given pivoting strategy is 
characterized by the growth factor

\begin{equation}
    \rho_n(\mat{A}) =  \frac{\max_{i,j,k} |a_{ij}^{(k)}|}{\max_{i,j} |a_{ij}|}
\end{equation}

\subsubsection{Wilkinson's backward error stability result for Gaussian elimination with partial pivoting}

Consider $\vec{\hat{x}}$ is the numerical solution to $\mat{A}\vec{x} = \vec{b}, \mat{A} \in \mathbb{R}^{n \times n}$
obtained using Gaussian elimination with partial pivoting. Then  $\vec{\hat{x}}$ is the
exact solution to the perturbed problem

\begin{equation}
    (\mat{A} + \mat{\Delta A}) \vec{\hat{x}} = \vec{b}, \quad ||\mat{\Delta A}||_\infty \leq n^2 \gamma_{3n} g_W(n) ||\mat{A}||_\infty
\end{equation}

where 

\begin{equation}
    ||\mat{A}||_\infty = \max_{1\leq i \leq n} \sum_{j=1}^{n} |a_{ij}| \quad \text{max row sum}
\end{equation}

and $\gamma_{3n}$ is defined in Eq. \ref{eq:mult_float_bound}. For partial pivoting,
$g_W(n) \leq 2^{n-1}$ (an exponential bound on the matrix size), in practice $g_W$
is of the order $n^{2/3}$ to $n^{1/2}$. For the details 
and proof see \cite[Theorem 9.5]{higham02}.

\subsubsection{LU factorization}
Let us ignore pivoting for now. We will

\begin{enumerate}
    \item rewrite Gauss elimination in matrix form
    \item see that Gauss elimination implicitly performs
    a matrix factorization $\mat{A} = \mat{L} \mat{U}$ with $L$ lower-diagonal
    and $U$ upper diagonal
\end{enumerate}


\subsubsubsection{Why are we interested in an LU factorization?}

Once we have a factorization of the form $\mat{A} = \mat{L} \mat{U}$, we can solve 
for any right-hand side $\vec{b}$ in $\mat{A}\vec{x} = \vec{b}$ based on
\begin{equation}
    \mat{A} \vec{x} = \mat{L} \mat{R} \vec{x} \quad \leftrightarrow \quad \mat{L} \vec{y} = \vec{b}, \quad \mat{R} \vec{x} = \vec{y}
\end{equation}
so by solving two triangular systems.

\begin{enumerate}
    \item (\textcolor{blue1}{Forward substitution}) Solve $\mat{L} \vec{y} = \vec{b}$ for $\vec{y}$. We can write the system as
    \begin{equation}
        \left(\begin{array}{ll}
        l_{11} & \\
        \vec{L}_{* 1} & \mat{L}_{* *}
        \end{array}\right)\left(\begin{array}{l}
        y_1 \\
        \vec{y}_*
        \end{array}\right)=\left(\begin{array}{l}
        b_1 \\
        \vec{b}_*
        \end{array}\right) \Rightarrow l_{11} y_1=b_1, \quad \vec{L}_{* 1} y_1+\mat{L}_{* *} \vec{y}_*=b_*
    \end{equation}
    from which we can solve for $y_1 = \frac{b_1}{l_{11}}$. We can then construct a new triangular
    system for $\vec{y}_*$
    \begin{equation}
        \mat{L}_{* *} \vec{y}_* = \vec{b}_* - \vec{L}_{* 1} y_1
    \end{equation}
    so $\vec{y}$ can be solved recursively, yielding the explicit formula
    \begin{equation}
        y_i=\frac{1}{l_{i i}}\left(b_i-\sum_{k=1}^{i-1} l_{i k} y_k\right)
    \end{equation}
    taking $\mathcal{O}(N^2)$ operations.
    \item (\textcolor{blue1}{Backward substitution}) Then $\mat{R} \vec{x} = \vec{y}$ for $\vec{x}$. The logic is the same as for forward substitution, 
    but bottom-up. We can write the system as
    \begin{equation}
        \left(\begin{array}{ll}
        \mat{R}_{* *} & \vec{R}_{* n} \\
        & r_{n n}
        \end{array}\right)\left(\begin{array}{l}
        \vec{x}_* \\
        x_1
        \end{array}\right)=\left(\begin{array}{l}
        \vec{y}_* \\
        y_n
        \end{array}\right)
    \end{equation}
    yielding the explicit formula
    \begin{equation}
        a_i=\frac{1}{r_{i i}}\left(y_i-\sum_{k=i+1}^N r_{i k} a_k\right)
    \end{equation}
    also taking $\mathcal{O}(N^2)$ operations.
\end{enumerate}

\subsubsubsection{Gauss elimination in matrix form I: Example}
We want to write our
Gaussian elimination update in matrix form.

Let us consider a specific example for Gaussian elimination

\begin{equation}
    \mat{A}^{(2)} = 
    \begin{pmatrix}
    a_{11}^{(2)} & a_{12}^{(2)} & a_{13}^{(2)} & a_{14}^{(2)} \\
    0            & a_{22}^{(2)} & a_{23}^{(2)} & a_{24}^{(2)} \\
    0            & a_{32}^{(2)} & a_{33}^{(2)} & a_{34}^{(2)} \\
    0            & a_{42}^{(2)} & a_{43}^{(2)} & a_{44}^{(2)}
    \end{pmatrix}
\end{equation}

so

\begin{equation}
    \mat{A}^{(3)} = 
    \begin{pmatrix}
    a_{11}^{(2)} & a_{12}^{(2)} & a_{13}^{(2)} & a_{14}^{(2)} \\
    0            & a_{22}^{(2)} & a_{23}^{(2)} & a_{24}^{(2)} \\
    0            & a_{32}^{(2)} - a_{22}^{(2)} \frac{a_{32}^{(2)}}{a_{22}^{(2)}} & a_{33}^{(2)} - a_{23}^{(2)} \frac{a_{32}^{(2)}}{a_{22}^{(2)}} & a_{34}^{(2)} - a_{24}^{(2)} \frac{a_{32}^{(2)}}{a_{22}^{(2)}} \\
    0            & a_{42}^{(2)} - a_{22}^{(2)} \frac{a_{42}^{(2)}}{a_{22}^{(2)}} & a_{43}^{(2)} - a_{23}^{(2)} \frac{a_{42}^{(2)}}{a_{22}^{(2)}} & a_{44}^{(2)} - a_{24}^{(2)} \frac{a_{42}^{(2)}}{a_{22}^{(2)}}
    \end{pmatrix}
\end{equation}

We plug in $m_{ik} = \frac{a^{(k)}_{ik}}{a^{(k)}_{kk}}$

\begin{equation}
    \begin{aligned}
    \mat{A}^{(3)} &= 
    \begin{pmatrix}
    a_{11}^{(2)} & a_{12}^{(2)} & a_{13}^{(2)} & a_{14}^{(2)} \\
    0            & a_{22}^{(2)} & a_{23}^{(2)} & a_{24}^{(2)} \\
    0            & a_{32}^{(2)} - a_{22}^{(2)} m_{32} & a_{33}^{(2)} - a_{23}^{(2)} m_{32} & a_{34}^{(2)} - a_{24}^{(2)} m_{32} \\
    0            & a_{42}^{(2)} - a_{22}^{(2)} m_{42} & a_{43}^{(2)} - a_{23}^{(2)} m_{42} & a_{44}^{(2)} - a_{24}^{(2)} m_{42}
    \end{pmatrix} \\
    &= \begin{pmatrix}
    a_{11}^{(2)} & a_{12}^{(2)} & a_{13}^{(2)} & a_{14}^{(2)} \\
    0            & a_{22}^{(2)} & a_{23}^{(2)} & a_{24}^{(2)} \\
    0            & a_{32}^{(2)} & a_{33}^{(2)} & a_{34}^{(2)} \\
    0            & a_{42}^{(2)} & a_{43}^{(2)} & a_{44}^{(2)}
    \end{pmatrix} -   
    \begin{pmatrix}
    0 & 0 & 0 & 0 \\
    0 & 0 & 0 & 0 \\
    0 & a_{22}^{(2)} m_{32} & a_{23}^{(2)} m_{32} & a_{24}^{(2)} m_{32} \\
    0 & a_{22}^{(2)} m_{42} & a_{23}^{(2)} m_{42} & a_{24}^{(2)} m_{42}
    \end{pmatrix} \\
    &= \begin{pmatrix}
    a_{11}^{(2)} & a_{12}^{(2)} & a_{13}^{(2)} & a_{14}^{(2)} \\
    0            & a_{22}^{(2)} & a_{23}^{(2)} & a_{24}^{(2)} \\
    0            & a_{32}^{(2)} & a_{33}^{(2)} & a_{34}^{(2)} \\
    0            & a_{42}^{(2)} & a_{43}^{(2)} & a_{44}^{(2)}
    \end{pmatrix}  -   
    \begin{pmatrix}
    0 & 0 & 0 & 0 \\
    0 & 0 & 0 & 0 \\
    0 & m_{32} & 0 & 0 \\
    0 & m_{42} & 0 & 0
    \end{pmatrix}
    \begin{pmatrix}
    a_{11}^{(2)} & a_{12}^{(2)} & a_{13}^{(2)} & a_{14}^{(2)} \\
    0            & a_{22}^{(2)} & a_{23}^{(2)} & a_{24}^{(2)} \\
    0            & a_{32}^{(2)} & a_{33}^{(2)} & a_{34}^{(2)} \\
    0            & a_{42}^{(2)} & a_{43}^{(2)} & a_{44}^{(2)}
    \end{pmatrix} \\
    &=
    \left( \mat{1} - \begin{pmatrix}
        0 \\ 0 \\ m_{32} \\ m_{42}
    \end{pmatrix} \begin{pmatrix}
        0 & 1 & 0 & 0
    \end{pmatrix} \right) \mat{A}^{(2)} \\
    &=
    \left( \mat{1} - \vec{m}_2 \vec{\hat{e}}_2^T \right) \mat{A}^{(2)}
    \end{aligned}
\end{equation}

This generalizes to

\begin{equation}
    \mat{A}^{(k+1)} = \mat{M}_k \mat{A}^{(k)}, \quad \mat{M}_k = (\mat{1} - \vec{m}_k \vec{e}_k^T), \quad (\vec{m}_k)_i = \begin{cases}
                0, &i \leq k \\
                \frac{a^{(k)}_{ik}}{a^{(k)}_{kk}}, &i>k
            \end{cases}
\end{equation}

The general relation is derived in the next section.

\subsubsubsection{Gauss elimination in matrix form II: Index notation}

We start by removing the bounds on $i$ and $j$ 
in our Gaussian elimination update

\begin{equation}
    a_{i j}^{(k+1)}=a_{i j}^{(k)}-m_{i k} a_{k j}^{(k)}, \quad i=k+1, ..., n, \quad j=k+1, ..., n
\end{equation}

as follows:

\begin{itemize}
    \item we start $j$ from $k$, making $a_{i>k,k}^{(k+1)} = 0$ explicit
    \item as for $j<k: a_{kj}^{(k)} = 0$, we can also start $j$ at $1$
    \item finally, by redefining
        \begin{equation}
            m_{ik} = \begin{cases}
                0, &i \leq k \\
                \frac{a^{(k)}_{ik}}{a^{(k)}_{kk}}, &i>k
            \end{cases}
        \end{equation}
    we can entirely remove the bounds on the indices.
\end{itemize}

For $i,j=1, ..., n$

\begin{equation}
    a_{ij}^{(k+1)} = a_{ij}^{(k)} - m_{ik} a_{kj}^{(k)} = a_{ij}^{(k)} - \underbrace{m_{ik} \delta_{kl}}_{=(\vec{m}_k \vec{\hat{e}}_k^T)_{il}} a_{lj}^{(k)}
\end{equation}

where $\vec{m}_k = (m_{1k},...,m_{nk})^T$ and $(\vec{\hat{e}}_k)_l = \delta_{kl}$ where $\delta_{kl}$ is the Kronecker delta.

\greybox{A matrix product $\mat{C} = \mat{A} \mat{B}$
corresponds to $c_{ij} = \sum_{l=1}^{n} a_{il} b_{lj} = a_{il} b_{lj}$
in \textbf{index notation}, where we implicitly sum over repeated indices.}

The corresponding matrix notation is

\begin{equation}
    \mat{A}^{(k+1)} = \mat{M}_k \mat{A}^{(k)}, \quad \mat{M}_k = \mat{1} - \vec{m}_k \vec{e}_k^T
\end{equation}

\subsubsubsection{From Gauss elimination in matrix form to LU factorization}

Gauss elimination results in an upper diagonal matrix

\begin{equation}
    \mat{U} := \mat{A}^{(n)} = \mat{M}_{n-1} \dots \mat{M}_{2} \mat{M}_{1} \mat{A}
\end{equation}

Note that

\begin{equation}
    \mat{M}_{k}^{-1} = \mat{1} + \vec{m}_k \vec{e}_k^T
\end{equation}

You can prove this by showing $\mat{M}_{k}^{-1} \mat{M}_{k} = \mat{1}$. You will need to use 
$(\vec{m}_k \vec{e}_k^T)(\vec{m}_k \vec{e}_k^T) = \mat{0}$ as $(\vec{m}_k)_{i \leq k} = 0$.

This naturally leads to the matrix factorization

\begin{equation}
    \begin{aligned}
        \mat{A}&= \mat{M}_{1}^{-1} \mat{M}_{2}^{-1} \dots \mat{M}_{n-1}^{-1} \mat{U} \\
               &= \left( 1 + \sum_{i = 1}^{n-1} \vec{m}_i \vec{\hat{e}}_i^T \right) \mat{U} \\
               &= \begin{pmatrix}
                1 \\
                m_{21} & 1 \\
                \vdots & m_{32} & \ddots \\
                \vdots & \vdots & & \ddots \\
                m_{n1} & m_{n2} & \dots & m_{n,n-1} & 1
               \end{pmatrix} \mat{U} \\
               &=: \mat{L} \mat{U} 
    \end{aligned}
\end{equation}

where $L$ is lower diagonal.

\bluebox{\textbf{Gauss elimination implicitly performs a factorization $\mat{A} = \mat{L} \mat{U}$ with $\mat{L}$ lower diagonal and $\mat{U}$ upper diagonal.}}

\subsubsubsection{Calculating the LU-decomposition in $\mathcal{O}(N^3)$}
From
\begin{equation}
    \left(\begin{array}{ll}
    a_{11} & \vec{A}^T_{1 *} \\
    \vec{A}_{* 1} & \mat{A}_{* *}
    \end{array}\right)=\left(\begin{array}{ll}
    l_{11} & \\
    \vec{L}_{* 1} & \mat{L}_{* *}
    \end{array}\right)\left(\begin{array}{ll}
    r_{11} & \vec{R}^T_{1 *} \\
    & \mat{R}_{* *}
    \end{array}\right), \quad \text { set diagonal of } \mat{L} \text { to } 1 \rightarrow \text { unique decomp }
\end{equation}
we can devise an $\mathcal{O}(N^3)$ algorithm to calculate $\mat{L}$ and $\mat{R}$,
see code \ref{code:lu_decomposition}.

\begin{codebox}[!htb]
    \begin{minted}{julia}
        # in-place decomposition of A into L and R, the upper-right part of A is overwritten by R,
        # the lower-left part of A is overwritten by L without diagonal (set to 1)
        function lu_decomposition!(A::Matrix{T}) where T <: Number
            N = size(A, 1)
            for i in 1:N
                for j in i+1:N
                    A[j, i] /= A[i, i]
                    for k in i+1:N
                        A[j, k] -= A[j, i] * A[i, k]
                    end
                end
            end
        end
        # one can then extract L = tril(A, -1) + I
        # and U = triu(A) via the LinearAlgebra package
    \end{minted}
    \caption{LU-decomposition of a matrix $\mat{A} \in \mathbb{R}^{N\times N}$ in $\mathcal{O}(N^3)$.}
    \label{code:lu_decomposition}
\end{codebox}

\note{Blocked versions can be parallelized, QR-decomposition retains the condition number of $\mat{A}$.}

\problem{LU-decomposition is $\mathcal{O}(N^3)$.}

\subsubsection{When does the LU factorization exist? \skipthis}

A unique LU factorization exists of $\mat{A} \in \mathbb{R}^{n \times n}$ if and only if
$\mat{A}_k := \mat{A}(1:k,1:k)$ is nonsingular for $k=1,...,n-1$. Otherwise
an LU factorization may still exist but if so not unique. See \cite[Theorem 9.2]{higham02} for a proof.

\subsection{Iterative methods}

\subsubsection{Jacobi iteration | a splitting method}
\bluebox{\textbf{Aim:} We want quicker than $\mathcal{O}(N^3)$, approximate solutions to linear systems.}

Consider the linear system
\begin{equation}
    \mat{A} \vec{x} = \vec{b}, \quad \mat{A} \in \mathbb{R}^{N\times N}, \vec{x}, \vec{b} \in \mathbb{R}^N
\end{equation}
For Jacobi iteration, we start with the trivial decomposition
\begin{equation}
    \begin{gathered}
        \mat{A} = \mat{D} - (\mat{L} + \mat{U}), \quad \text{diagonal part } \mat{D}, \quad \text{negative left-below-diagonal part} \mat{L} \\
        \text{negative right-above-diagonal part} \mat{U}
    \end{gathered}
\end{equation}
so
\begin{equation}
    \mat{A}\vec{x} = \left[ \mat{D} - (\mat{L} + \mat{U}) \right] \vec{x} = \vec{b} \quad \rightarrow \quad \vec{x} = \mat{D}^{-1} \vec{b} + \mat{D}^{-1} (\mat{L} + \mat{U}) \vec{x}
\end{equation}
so $\vec{x}$ is a fixed point of the RHS, leading to fixed-point, here \textcolor{blue1}{Jacobi iteration}
\begin{equation}
    \boxed{\vec{x}^{(n+1)} = \mat{D}^{-1} \vec{b} + \mat{D}^{-1} (\mat{L} + \mat{U}) \vec{x}^{(n)}, \quad (\mat{D}^{-1})_{ii} = \frac{1}{A_{ii}}}
\end{equation}

\subsubsubsection{When does the Jacobi iteration converge?}
The Jacobi iteration converges if and only if all eigenvalues $\lambda_i$ of the convergence
matrix
\begin{equation}
    \mat{M} := \mat{D}^{-1} (\mat{L} + \mat{U})
\end{equation}
are smaller than one.
\begin{equation}
    \text{spectral radius } \rho_s(\mat{M}) \equiv \max_i{|\lambda_i|} < 1
\end{equation}
\greenbox{The smaller the spectral radius, the faster the convergence.}
\paragraph{Derivation of the convergence criterion}
Consider the error at step $n+1$
\begin{equation}
    \begin{gathered}
        \vec{e}^{(n+1)} = \vec{x}_{\text{exact}} - \vec{x}^{(n+1)} \underbrace{=}_{\text{Jacobi-Iteration}} \vec{x}_{\text{exact}} - \left( \mat{D}^{-1} \vec{b} + \mat{M} \vec{x}^{(n)} \right) \\
    \underbrace{=}_{\vec{x}_{\text{exact}} = \mat{D}^{-1}\vec{b}+\mat{M}\vec{x}_{\text{exact}}} \mat{M} (\vec{x}_{\text{exact}} - \vec{x}^{(n)}) = \mat{M} \vec{e}^{(n)}
    \end{gathered}
\end{equation}

so
\begin{equation}
    \vec{e}^{(n)} = \mat{M}^n \vec{e}^{(0)}
\end{equation}
which scales with $\rho_s^n(\mat{M})$ for $n$ sufficiently large (follows from decomposing $\vec{e}^{(0)}$ into eigenvectors of $\mat{M}$)
so the convergence criterion follows.

\problem{The convergence is usually slow, better use Gauss-Seidel iteration.}

\subsubsubsection{Example Jacobi step}
Consider
\begin{equation}
    \mat{A} \vec{x} = \begin{pmatrix} 10 & -5 & 0 \\ -5 & 10 & -5 \\ 0 & -5 & 5 \end{pmatrix} \left( \begin{array}{c} x_1 \\ x_2 \\ x_3 \end{array} \right) = \vec{b} = \left( \begin{array}{c} 0 \\ 0 \\ 15 \end{array} \right)
\end{equation}
then
\begin{equation}
    \mat{D} = \begin{pmatrix} 10 & 0 & 0 \\ 0 & 10 & 0 \\ 0 & 0 & 5 \end{pmatrix}, \quad \mat{L} = \begin{pmatrix} 0 & 0 & 0 \\ 5 & 0 & 0 \\ 0 & 5 & 0 \end{pmatrix}, \quad \mat{R} = \begin{pmatrix} 0 & 5 & 0 \\ 0 & 0 & 5 \\ 0 & 0 & 0 \end{pmatrix}
\end{equation}
so
\begin{equation}
    \vec{x}^{(0)} = \left( \begin{array}{c} 0 \\ 0 \\ 0 \end{array} \right) \quad \rightarrow \quad \vec{x}^{(1)} = \left( \begin{array}{c} \frac{1}{10} b_0 + \frac{1}{10}\cdot 5 x^{(0)}_2 \\ \frac{1}{10} b_1 + \frac{1}{10}\cdot \left( 5 x^{(0)}_1 + 5 x^{(0)}_3 \right) \\ \frac{1}{5} b_1 + \frac{1}{5} \cdot 5 x^{0}_2 \end{array} \right)
\end{equation}
