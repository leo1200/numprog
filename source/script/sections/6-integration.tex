\section{Integration}
\thispagestyle{plain}

Consider a function $f: [a,b] \to \mathbb{R}$. We are
interested in the integral

\begin{equation}
    \int_a^b f(x) \, dx
\end{equation}

In general, there might not be an analytical
solution to the integral which can be written
in elementary form, as is the case for
$f(x) = \exp{(-x^2)}$. So how can we calculate
integrals numerically?

Monte Carlo integration,
advantageous for high-dimensional integration,
is covered in Sec. \ref{app:monte_carlo}. Additionally,
notes on Gauss-Legendre quadrature are provided in Sec. \ref{app:gauss_legendre}.

\subsection{Classical one-dimensional quadrature}

\greenbox{\textbf{Idea:} On small subsection, we can faithfully
approximate $f(x)$ by simpler functions with analytical integrals.}

We start by dividing the domain $[a,b]$ into sub-intervals
$[x_i,x_{i+1}], x_0 = a, x_N = b$, such that

\begin{equation}
    \int_a^b f(x) \, dx = \sum_{i=0}^{N-1} \int_{x_i}^{x_{i+1}} f(x) \, dx
\end{equation}

Based on the approximation of $f(x)$ on these sub-intervals, different
numerical integration schemes come up:

\begin{itemize}
    \item \textbf{Midpoint rule}: For the central-constant approximation
    \begin{equation}
        f(x) \approx f\left( \frac{x_{i} + x_{i+1}}{2} \right) \quad \text{for} \quad x \in [x_i, x_{i+1}]
    \end{equation}
    we obtain the midpoint rule
    \begin{equation}
        \int_{x_i}^{x_{i+1}} f(x) \, dx = \underbrace{(x_{i+1} - x_i) \, f\left( \frac{x_{i} + x_{i+1}}{2} \right)}_{=:M_i} + \mathcal{O}((x_{i+1} - x_i)^3)
    \end{equation}
    \item \textbf{Trapezoidal rule}: For the linear approximation
    \begin{equation}
        f(x) \approx f\left( x_{i} \right) + \frac{x-x_{i}}{x_{i+1}-x_{i}} \, \left( f\left( x_{i+1} \right) - f\left( x_{i} \right) \right) \quad \text{for} \quad x \in [x_i, x_{i+1}]
    \end{equation}
    we obtain the trapezoidal rule
    \begin{equation}
        \int_{x_i}^{x_{i+1}} f(x) \, dx = \underbrace{(x_{i+1} - x_i) \, \left(\frac{f(x_i) + f(x_{i+1})}{2}\right)}_{=:T_i} + \mathcal{O}((x_{i+1} - x_i)^3)
    \end{equation}
    \item \textbf{Simpson's rule}: We interpolate $f(x)$ through $\left(x_{i}, f(x_{i})\right), \left(\frac{x_{i} + x_{i+1}}{2}, f\left(\frac{x_{i} + x_{i+1}}{2}\right)\right), \left(x_{i+1}, f(x_{i+1})\right)$
    with a polynomial
    \begin{equation}
        f(x) \approx a(x-x_i)^2 + b(x-x_i) + c \quad \text{for} \quad x \in [x_i, x_{i+1}]
    \end{equation}
    yielding (we leave the derivation as a task to the reader)
    \begin{equation}
        \int_{x_i}^{x_{i+1}} f(x) \, dx = \underbrace{\frac{x_{i+1} - x_i}{6} \, \left(f(x_i) + 4 f\left(\frac{x_i + x_{i+1}}{2}\right) + f(x_{i+1})\right)}_{=:S_i} + \mathcal{O}((x_{i+1}-x_i)^5)
    \end{equation}
\end{itemize}

\subsubsection{Local error derivations}

We can use Taylor expansion to derive error terms for quadrature rules.

Let us define

\begin{equation}
    c_i := \frac{x_{i} + x_{i+1}}{2}, \quad \Delta x := x_{i+1} - x_i
\end{equation}

and consider the \textbf{midpoint approximation}

\begin{equation}
    f(x) = f\left( c_i \right) + (x-c_i) f'(c_i) + \frac{1}{2}(x-c_i)^2 f''(c_i) + \mathcal{O}((x-c_i)^3) \quad \text{for} \quad x \in [x_i, x_{i+1}]
\end{equation}

yielding

\begin{equation}
    \begin{aligned}
        \int_{x_i}^{x_{i+1}} f(x) \, dx &= \Delta x f\left( c_i \right) + \underbrace{\int_{x_i}^{x_{i+1}} (x-c_i) f'(c_i)  \, dx}_{=0} + \int_{x_i}^{x_{i+1}} \frac{1}{2}(x-c_i)^2 f''(c_i)  \, dx + \mathcal{O}(\Delta x^4) \\
        &= \Delta x f\left( c_i \right) + 2 f''(c_i) \int_{c_i}^{x_{i+1}} \frac{1}{2}(x-c_i)^2   \, dx + \mathcal{O}(\Delta x^4) \\
        &= \Delta x f\left( c_i \right) + \frac{1}{24} f''(c_i)\Delta x^3 + \mathcal{O}(\Delta x^4)
    \end{aligned}
\end{equation}

such that

\begin{equation}
    M_i = M_{[x_i, x_{i+1}]}(f) := \Delta x f\left( c_i \right) = \int_{x_i}^{x_{i+1}} f(x) \, dx - \frac{1}{24} f''(x_i)\Delta x^3 + \mathcal{O}(\Delta x^4)
\end{equation}

where we have used $f''(c_i) = f''(x_i) + \mathcal{O}(\Delta x)$.

Likewise, for the trapezoidal rule, we can find

\begin{equation}
    T_i = T_{[x_i, x_{i+1}]}(f) := \Delta x \frac{f(x_i) + f(x_{i+1})}{2} = \int_{x_i}^{x_{i+1}} f(x) \, dx + \frac{1}{12} f''(x_i)\Delta x^3 + \mathcal{O}(\Delta x^4)
\end{equation}

\greenbox{\textbf{Idea:} We can combine $M_i$ and $T_i$ to cancel the third order error terms.}

\begin{equation}
    S_i = S_{[x_i, x_{i+1}]}(f) := \frac{2M+T}{3} = \frac{x_{i+1} - x_i}{6} \, \left(f(x_i) + 4 f\left(\frac{x_i + x_{i+1}}{2}\right) + f(x_{i+1})\right) = \int_{x_i}^{x_{i+1}} f(x) \, dx + \mathcal{O}(\Delta x^4)
\end{equation}

\textbf{We have just re-derived Simpson's rule!} But we previously stated the error term $\mathcal{O}(\Delta x^5)$.
We will see that the even error terms vanish by symmetry.

Let us split $f$ into an even and odd part around $c_i$:

\begin{equation}
    f(x) = f_e(x) + f_o(x), \quad f_e(c_i - x) = f_e(c_i + x), \quad  f_o(c_i - x) = -f_o(c_i + x)
\end{equation}

By symmetry

\begin{equation}
    S_i(f_o) = \int_{x_i}^{x_{i+1}} f_o(x) \, dx = 0, \quad c_i = \frac{x_i + x_{i+1}}{2}
\end{equation}

i.e. the uneven part is integrated exactly.

The Taylor expansion of $f_e$ has only even terms, such that
an approximation to $\int_{x_i}^{x_{i+1}} f_e(x) \, dx$ based on truncated
Taylor expansion can only have uneven error terms. Therefore

\begin{equation}
    S_i =\frac{x_{i+1} - x_i}{6} \, \left(f(x_i) + 4 f\left(\frac{x_i + x_{i+1}}{2}\right) + f(x_{i+1})\right) = \int_{x_i}^{x_{i+1}} f(x) \, dx + \mathcal{O}(\Delta x^5)
\end{equation}

For a proof on the specific error term

\begin{equation}
    E_\text{local} = \int_{x_i}^{x_{i+1}} (f(x) - q_i(x)) \, dx = -\frac{f^{(4)}(\xi)}{32 \cdot 90} (\underbrace{x_{i+1} - x_i}_{=: \Delta x})^5 \sim \mathcal{O}(\Delta x^5)
\end{equation}

(with $q_i$ the polynomial that agrees with $f$ at $x_i, \frac{x_i + x_{i+1}}{2}, x_{i+1}$) see \cite{simpson_error}.

\subsubsection{Global errors}

Consider a constant sub-interval size $x_{i+1} - x_i = \Delta x$. The whole
domain of integration $[a,b]$ is divided into $N = \frac{b-a}{\Delta x}$ sub-intervals.
Therefore, when the local error scales with $\mathcal{O}(\Delta x^n)$, the global errors
scales with $N \mathcal{O}(\Delta x^n) =  \mathcal{O}(\Delta x^{n-1})$.
