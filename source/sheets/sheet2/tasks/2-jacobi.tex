\problem{Jacobi iteration}

    Consider random symmetric strictly diagonally dominant matrices
    $\mat{R}_N$ with 
    \begin{equation}
        \forall i: |r_{ii}| > \sum_{j \neq i} |a_{ij}| 
    \end{equation}
    and $\mat{R}_N^T = \mat{R}_N$. These can be generated by
    \begin{enumerate}
        \item Draw a random Gaussian matrix $\mat{G}_N \in \mathbb{R}^{n \times n}, G_{ij} \sim \mathcal{N}(\mu = 0,\sigma^2 = 1)$.
        \item Enforce symmetry by $\mat{S}_N = \frac{1}{2} (\mat{G}_N + \mat{G}_N^T)$.
        \item Enforce strict diagonal dominance by
        \begin{equation}
            R_{ij} = \begin{cases}
                S_{ij}, \quad &i \neq j, \\
                \delta + \sum_{j \neq i} |S_{ij}|, \quad &i = j
            \end{cases}
        \end{equation}
    with margin $\delta = 0.1$.
    \end{enumerate}

    A Gaussian RHS in $\mat{A}\vec{x} = \vec{b}$, $b_i \sim \mathcal{N}(\mu = 0,\sigma^2 = 1), i = 1,...,N$ is used throughout this task.

    \bluebox{\textbf{Jacobi iteration:} For Jacobi iteration, we start with the trivial decomposition
    \begin{equation}
        \begin{gathered}
            \mat{A} = \mat{D} - (\mat{L} + \mat{U}), \quad \text{diagonal part } \mat{D}, \quad \text{negative left-below-diagonal part} \mat{L} \\
            \text{negative right-above-diagonal part} \mat{U}
        \end{gathered}
    \end{equation}
    and iterate
    \begin{equation}
        \vec{x}^{(n+1)} = \mat{D}^{-1} \vec{b} + \underbrace{\mat{D}^{-1} (\mat{L} + \mat{U})}_{=:\mat{M}} \vec{x}^{(n)}, \quad (\mat{D}^{-1})_{ii} = \frac{1}{A_{ii}}
    \end{equation}
    You can start the iteration with $\vec{x}^{(0)} = \vec{0}$.
    }

    \begin{enumerate}
        \item Show that Jacobi iteration for a strictly diagonally dominant matrix converges, i.e. 
        the spectral radius $\rho_s(\mat{M}) < 1$.
        \item Implement the generation of $\mat{R}_N$ and $\vec{b}\in\mathbb{R}^N$ with $b_i \sim \mathcal{N}(\mu = 0,\sigma^2 = 1), i = 1,...,N$ in Python.
        \item Implement the Jacobi iteration in Python. For $N = 5, 50, 150$ plot the \textit{normwise backward error}
        \begin{equation}
            \eta^{(n)} = \frac{||\vec{r}^{(n)}||_1}{||\mat{A}||_1 ||\vec{\hat{x}}^{(n)}||_1 + ||\vec{b}||_1}, \quad \vec{r} = \mat{A}\vec{\hat{x}}^{(n)} - \vec{b}
        \end{equation}
        for $n = 0, 1, ..., N$.
    \end{enumerate}
