\problem{Jacobi iteration}

    Consider random strictly diagonally dominant matrices
    $\mat{R}_N$ with 
    \begin{equation}
        \forall i: |r_{ii}| > \sum_{j \neq i} |a_{ij}| 
    \end{equation}
    These can be generated by
    \begin{enumerate}
        \item Draw a random Gaussian matrix $\mat{G}_N \in \mathbb{R}^{n \times n}, G_{ij} \sim \mathcal{N}(\mu = 0,\sigma^2 = 1)$.
        \item Enforce strict diagonal dominance by
        \begin{equation}
            R_{ij} = \begin{cases}
                G_{ij}, \quad &i \neq j, \\
                \delta + \sum_{j \neq i} |G_{ij}|, \quad &i = j
            \end{cases}
        \end{equation}
    with margin $\delta = 0.1$.
    \end{enumerate}

    A Gaussian RHS in $\mat{A}\vec{x} = \vec{b}$, $b_i \sim \mathcal{N}(\mu = 0,\sigma^2 = 1), i = 1,...,N$ is used throughout this task.

    \bluebox{\textbf{Jacobi iteration:} For Jacobi iteration, we start with the trivial decomposition
    \begin{equation}
        \begin{gathered}
            \mat{A} = \mat{D} - (\mat{L} + \mat{U}), \quad \text{diagonal part } \mat{D}, \quad \text{negative left-below-diagonal part} \mat{L} \\
            \text{negative right-above-diagonal part} \mat{U}
        \end{gathered}
    \end{equation}
    and iterate
    \begin{equation}
        \vec{x}^{(n+1)} = \mat{D}^{-1} \vec{b} + \underbrace{\mat{D}^{-1} (\mat{L} + \mat{U})}_{=:\mat{M}} \vec{x}^{(n)}, \quad (\mat{D}^{-1})_{ii} = \frac{1}{A_{ii}}
    \end{equation}
    You can start the iteration with $\vec{x}^{(0)} = \vec{0}$.
    }

    \begin{enumerate}
        \item Show that Jacobi iteration converges for a strictly diagonally dominant matrix, i.e. 
        the spectral radius $\rho_s(\mat{M}) < 1$.
        \item Implement the generation of $\mat{R}_N$ and $\vec{b}\in\mathbb{R}^N$ with $b_i \sim \mathcal{N}(\mu = 0,\sigma^2 = 1), i = 1,...,N$ in Python.
        \item Implement the Jacobi iteration in Python. For $N = 5, 50, 150$ plot the \textit{normwise backward error}
        \begin{equation}
            \eta^{(n)} = \frac{||\vec{r}^{(n)}||_1}{||\mat{A}||_1 ||\vec{\hat{x}}^{(n)}||_1 + ||\vec{b}||_1}, \quad \vec{r} = \mat{A}\vec{\hat{x}}^{(n)} - \vec{b}
        \end{equation}
        for $n = 0, 1, ..., N$.
    \end{enumerate}


\begin{solutionblock}
    \part
    The Jacobi iteration matrix is $\mat{M} = \mat{D}^{-1}(\mat{L} + \mat{U})$. The components of $\mat{M}$ are $M_{ij} = -A_{ij}/A_{ii}$ for $i \neq j$ and $M_{ii} = 0$. 
    Convergence is guaranteed if the spectral radius $\rho_s(\mat{M}) < 1$. Using the induced $\infty$-norm (maximum row sum):
    \begin{equation}
        \|\mat{M}\|_\infty = \max_i \sum_{j=1}^N |M_{ij}| = \max_i \sum_{j \neq i} \frac{|A_{ij}|}{|A_{ii}|}
    \end{equation}
    Since $\mat{A}$ is strictly diagonally dominant, $|A_{ii}| > \sum_{j \neq i} |A_{ij}|$ for all $i$. This implies $\frac{\sum_{j \neq i} |A_{ij}|}{|A_{ii}|} < 1$, so $\|\mat{M}\|_\infty < 1$. Since $\rho_s(\mat{M}) \leq \|\mat{M}\|_\infty$, the iteration must converge.

    \part
    The system generation is given in Code-Snipped \ref{code:gen_system}.
    \begin{codebox}
        \begin{minted}{python3}
        def generate_system(N, delta=0.1):
            G = np.random.normal(0, 1, (N, N))
            R = G.copy()
            for i in range(N):
                # Sum of absolute values of non-diagonal elements
                off_diag_sum = np.sum(np.abs(G[i, :])) - np.abs(G[i, i])
                R[i, i] = delta + off_diag_sum
            b = np.random.normal(0, 1, N)
            return R, b
        \end{minted}
        \caption{Generation of strictly diagonally dominant matrix $\mat{R}_N$.}
        \label{code:gen_system}
    \end{codebox}

    \part
    \begin{codebox}
        The Jacobi solver is given in Code-Snippet \ref{code:jacobi}.
        \begin{minted}{python3}
        def jacobi_solve(A, b, max_iter):
            N = len(b)
            x = np.zeros(N)
            D = np.diag(A)
            LU = np.diag(D) - A # Matrix (L+U)
            
            norm_A = np.linalg.norm(A, 1)
            norm_b = np.linalg.norm(b, 1)
            errors = []

            for n in range(max_iter + 1):
                # Calculate normwise backward error
                r = A @ x - b
                eta = np.linalg.norm(r, 1) / (norm_A * np.linalg.norm(x, 1) + norm_b)
                errors.append(eta)
                
                # Jacobi Update
                if n < max_iter:
                    x = (b + LU @ x) / D
            return errors

        # Plotting logic
        for N in [5, 50, 150]:
            A, b = generate_system(N)
            errs = jacobi_solve(A, b, N)
            plt.semilogy(range(N + 1), errs, label=f'N={N}')
        plt.legend(); plt.grid(True); plt.show()
        \end{minted}
        \caption{Jacobi implementation and error tracking.}
        \label{code:jacobi}
    \end{codebox}
\end{solutionblock}
